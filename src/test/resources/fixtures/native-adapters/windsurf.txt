== hook KEEP
#!/bin/sh
# sentinel-owned: windsurf-native-hooks v1
# Native event adapter for windsurf. Usage: sentinel.sh pre|post, with the event JSON on stdin.
# Hooks may run in a subdirectory, so Sentinel always checks the project this file is in.
cd "$(dirname "$0")/../.." || exit 2
case "$1" in
  pre)
    [ "keep" = "enable" ] || exit 0
    exec sentinel hook native --agent windsurf --guard-approved-specs --event-stdin ;;
  post) exec sentinel hook native --agent windsurf --event-stdin ;;
  *) echo "Sentinel hook wrapper: expected pre or post." >&2; exit 2 ;;
esac
